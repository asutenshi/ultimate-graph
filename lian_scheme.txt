@startuml
title Алгоритм LIAN
skinparam conditionStyle diamond
skinparam backgroundColor #FDFDFD

start
:Инициализация;
if (Start или Goal невалидны?) then (да)
    #Pink:Вернуть пустой путь;
    stop
else (нет)
endif

:OpenSet.push(StartNode);
:CostSoFar[Start] = 0;

while (OpenSet не пуст?) is (да)
    :Current = OpenSet.pop();
    
    if (Current.cost > CostSoFar[Current.pos]?) then (да)
        :Пропустить (устаревший узел);
    else (нет)
        
        partition "Проверка угла (Lazy Check)" {
            if (Угол(GrandParent -> Parent -> Current) > MaxAngle?) then (да)
                :Пропустить (угол слишком резкий);
            else (нет)
            
                if (Current == Goal?) then (да)
                    #LightGreen:Восстановить путь\n(reconstructPath);
                    stop
                endif

                partition "Оптимизация прямой видимости" {
                    if (Прямая линия(Current, Goal) свободна\nИ Угол валиден?) then (да)
                        :Добавить прямую линию к пути;
                        #LightGreen:Вернуть путь;
                        stop
                    endif
                }

                :Получить соседей (getNeighbors)\n(по кругу радиуса Delta);
                
                while (Есть еще соседи?) is (да)
                    :Next = следующий сосед;
                    
                    partition "Логика срезания углов (Shortcut)" {
                        if (Линия(Parent, Next) свободна\nИ Угол валиден?) then (да)
                            :Parent = Current.Prev\n(Срезаем через родителя);
                        else (нет)
                            :Parent = Current.Pos\n(Идем от текущей);
                        endif
                    }
                    
                    :NewCost = CostToParent + Distance(Parent, Next);
                    
                    if (NewCost < CostSoFar[Next]?) then (да)
                        :CostSoFar[Next] = NewCost;
                        :Priority = NewCost + Heuristic(Next, Goal);
                        :OpenSet.push(Next);
                        :CameFrom[Next] = Parent;
                    endif
                endwhile (нет)
            
            endif
        }
    endif
endwhile (нет)

#Pink:Путь не найден;
stop

@enduml
